1. The button sits in **`TaskSidebar`'s bottom-left toolbar group**, directly after the notification bell (`components/sidebar/TaskSidebar.tsx`, the `<Toolbar noBorder …>` block below `TaskDropZone`, around lines 476-500). **Design.**
- **State.**
  - `stores/dev-server-store.ts` keeps one slice per backend through `createSlices<DevServer>()` (`lib/backend-scope.ts`) and drops a backend's slice and stop states in a `registerBackendReset` hook.
  - A `DEV_SERVER_LIST` broadcast replaces that backend's slice through `slices.apply(backendId, () => list)`. A replacing write is recorded in `pending`, so when a fetch is in flight, `replace()` replays it over the older response and the newer broadcast wins. Don't use `slices.replace`/`begin` for broadcasts.
- **Stop state.**
  - It is held per record in `stopStates`, keyed `${backendId}:${id}` (`devServerKey`). Each entry is `"stopping"` or `{ error: string }`; an absent key means idle.
  - A new attempt overwrites an error with `"stopping"`, and closing the popover clears errors (`clearStopErrors`).
  - A rejected request (detached backend, timeout) shows its `Error.message`.
  - On `{ ok: true }` the store also removes the row from its slice. The backend's broadcast does the same, and the write is idempotent; this way an `ok` for a record the backend no longer held still clears the row.
- **Fetch on attach.** `bootstrapBackend` (`stores/backend-store.ts`, about line 391) gets `fetchDevServers(id)` in its non-critical `Promise.allSettled`, beside `fetchNotifications`. An older backend that doesn't know the message rejects, and `allSettled` swallows it.
- **Components** (`components/sidebar/DevServerPopover.tsx`):
  - `DevServerButton` owns the popover's open state. It renders nothing when the count is 0 and the popover is closed, and its badge uses the success colour.
  - `DevServerList` is the popover body. It's exported for the component test, and it reads the stores itself.
  - The group title is `[<machine>] · <project> [· <task>]`. The machine name appears only when the rows come from more than one backend: `MachineState.displayName`, falling back to `backendId`. The project name falls back to `projectId`, as `NotificationPopover.getProjectName` does.
- **Shared helper.** `formatRelativeTime` moves out of `NotificationPopover.tsx` into `lib/relative-time.ts`, and both popovers import it. `ScheduleManagementDialog` has a different variant (nullable input), so leave it alone.
- **Zustand.** Use selectors with stable defaults and never destructure a whole store (see memory `project_zustand_reactivity.md`).
- **Not included:** no confirmation dialog, no navigation from a row, no TUI.

```ts
import { create } from "zustand";
import type {
    DevServer,
    DevServerListResponse,
    DevServerStopPayload,
    DevServerStopResult,
} from "@taskflow/shared";
import { MSG } from "@taskflow/shared";
import { createSlices, type Scoped } from "@/lib/backend-scope";
import { onEvent, sendRequest } from "@/lib/connection-registry";
import { registerBackendReset } from "./store-reset";

/** Absent from `stopStates`: idle. */
type DevServerStopState = "stopping" | { error: string };

interface DevServerStoreState {
    devServers: Scoped<DevServer>[];
    /** Keyed by `devServerKey`. */
    stopStates: Record<string, DevServerStopState>;

    fetchDevServers(backendId: string): Promise<void>;
    stopDevServer(server: Scoped<DevServer>): Promise<void>;
    /** Errors only: a stop still in flight keeps its "stopping". */
    clearStopErrors(): void;
}

const slices = createSlices<DevServer>();

function devServerKey(server: Pick<Scoped<DevServer>, "backendId" | "id">): string {
    return `${server.backendId}:${server.id}`;
}

function publish(): void {
    useDevServerStore.setState({ devServers: slices.read() });
}

function setStopState(key: string, state: DevServerStopState | null): void {
    useDevServerStore.setState((s) => {
        const rest = Object.fromEntries(Object.entries(s.stopStates).filter(([k]) => k !== key));
        return { stopStates: state === null ? rest : { ...rest, [key]: state } };
    });
}

const useDevServerStore = create<DevServerStoreState>((set, get) => ({
    devServers: [],
    stopStates: {},

    async fetchDevServers(backendId) {
        const landed = await slices.load(backendId, async () => {
            const { devServers } = await sendRequest<DevServerListResponse>(
                backendId,
                MSG.DEV_SERVER_LIST,
                {},
            );
            return devServers;
        });
        if (landed) publish();
    },

    async stopDevServer(server) {
        const key = devServerKey(server);
        if (get().stopStates[key] === "stopping") return;
        setStopState(key, "stopping");
        const payload: DevServerStopPayload = { id: server.id };
        let result: DevServerStopResult;
        try {
            result = await sendRequest<DevServerStopResult>(
                server.backendId,
                MSG.DEV_SERVER_STOP,
                payload,
            );
        } catch (error) {
            setStopState(key, { error: error instanceof Error ? error.message : String(error) });
            return;
        }
        if (!result.ok) {
            setStopState(key, { error: result.error });
            return;
        }
        // The backend's broadcast removes it too; this covers an `ok` for a record it no
        // longer held. Skip a machine reset meanwhile: `apply` would recreate its slice.
        if (slices.backends().includes(server.backendId)) {
            slices.apply(server.backendId, (items) => items.filter((d) => d.id !== server.id));
            publish();
        }
        setStopState(key, null);
    },

    clearStopErrors() {
        set((s) => ({
            stopStates: Object.fromEntries(
                Object.entries(s.stopStates).filter(([, state]) => state === "stopping"),
            ),
        }));
    },
}));

registerBackendReset("dev-server-store", (backendId) => {
    slices.drop(backendId);
    publish();
    const prefix = `${backendId}:`;
    useDevServerStore.setState((s) => ({
        stopStates: Object.fromEntries(
            Object.entries(s.stopStates).filter(([key]) => !key.startsWith(prefix)),
        ),
    }));
});

// A full list, so the write replaces the slice. Through `apply`, not `replace`: a
// fetch in flight replays it over its older response instead of overwriting it.
const _unsubDevServerList = onEvent(MSG.DEV_SERVER_LIST, (payload, backendId) => {
    const { devServers } = payload as DevServerListResponse;
    if (!Array.isArray(devServers)) return;
    const scoped = devServers.map((d) => ({ ...d, backendId }));
    slices.apply(backendId, () => scoped);
    publish();
});

export { useDevServerStore, devServerKey };
export type { DevServerStopState };

if (import.meta.hot) {
    import.meta.hot.dispose(() => {
        _unsubDevServerList();
    });
}
```

```tsx
    cleanups.push(() => {
        act(() => root.unmount());
        container.remove();
    });
    return container;
}

function groupTitles(container: HTMLElement): string[] {

    const row = [...container.querySelectorAll("[data-dev-server-row]")].find((el) =>
        el.textContent?.includes(label),
    );
    const button = row?.querySelector("button");
    if (!button) throw new Error(`no stop button for ${label}`);
    return button;
}
```

```ts
export function formatRelativeTime(dateStr: string): string {
    const seconds = Math.floor((Date.now() - new Date(dateStr).getTime()) / 1000);
    if (seconds < 60) return "just now";
    const minutes = Math.floor(seconds / 60);
    if (minutes < 60) return `${minutes}m ago`;
    const hours = Math.floor(minutes / 60);
    if (hours < 24) return `${hours}h ago`;
    const days = Math.floor(hours / 24);
    return `${days}d ago`;
}
```

`packages/ui/src/components/sidebar/DevServerPopover.tsx`:

```tsx
import { useCallback, useMemo, useState } from "react";
import { Server } from "lucide-react";
import type { DevServer, Project, Task } from "@taskflow/shared";
import type { Scoped } from "@/lib/backend-scope";
import { formatRelativeTime } from "@/lib/relative-time";
import { useBackendStore } from "@/stores/backend-store";
import type { MachineState } from "@/stores/backend-store";
import { devServerKey, useDevServerStore } from "@/stores/dev-server-store";
import { useProjectStore } from "@/stores/project-store";
import { useTaskStore } from "@/stores/task-store";
import { Button } from "../ui/button";
import { Popover, PopoverContent, PopoverTrigger } from "../ui/popover";

interface DevServerGroup {
    key: string;
    title: string;
    servers: Scoped<DevServer>[];
}

interface GroupSources {
    projects: Scoped<Project>[];
    tasks: Scoped<Task>[];
    machines: MachineState[];
}

function groupDevServers(
    devServers: Scoped<DevServer>[],
    { projects, tasks, machines }: GroupSources,
): DevServerGroup[] {
    const multiBackend = new Set(devServers.map((d) => d.backendId)).size > 1;
    const sorted = [...devServers].sort(
        (a, b) => new Date(a.registeredAt).getTime() - new Date(b.registeredAt).getTime(),
    );
    const groups = new Map<string, DevServerGroup>();
    for (const server of sorted) {
        const key = `${server.backendId}:${server.projectId}:${server.taskId ?? ""}`;
        let group = groups.get(key);
        if (!group) {
            const parts: string[] = [];
            if (multiBackend) {
                parts.push(
                    machines.find((m) => m.id === server.backendId)?.displayName ??
                        server.backendId,
                );
            }
            parts.push(
                projects.find((p) => p.backendId === server.backendId && p.id === server.projectId)
                    ?.name ?? server.projectId,
            );
            const task = server.taskId
                ? tasks.find((t) => t.backendId === server.backendId && t.id === server.taskId)
                : undefined;
            if (task) parts.push(task.title || task.description || "Untitled");
            group = { key, title: parts.join(" · "), servers: [] };
            groups.set(key, group);
        }
        group.servers.push(server);
    }
    return [...groups.values()];
}

function DevServerList() {
    const devServers = useDevServerStore((s) => s.devServers);
    const stopStates = useDevServerStore((s) => s.stopStates);
    const stopDevServer = useDevServerStore((s) => s.stopDevServer);
    const projects = useProjectStore((s) => s.projects);
    const liveTasks = useTaskStore((s) => s.tasks);
    const archivedTasks = useTaskStore((s) => s.archivedTasks);
    const machines = useBackendStore((s) => s.machines);

    const groups = useMemo(
        () =>
            groupDevServers(devServers, {
                projects,
                tasks: [...liveTasks, ...archivedTasks],
                machines,
            }),
        [devServers, projects, liveTasks, archivedTasks, machines],
    );

    if (groups.length === 0) {
        return <div className="text-muted-foreground px-3 py-4 text-sm">No dev servers running</div>;
    }

    return (
        <div className="max-h-80 overflow-y-auto py-1">
            {groups.map((group) => (
                <div key={group.key}>
                    <div
                        data-dev-server-group=""
                        className="text-muted-foreground truncate px-3 pt-2 pb-1 text-xs">
                        {group.title}
                    </div>
                    <ul>
                        {group.servers.map((server) => {
                            const state = stopStates[devServerKey(server)];
                            const stopping = state === "stopping";
                            const error = typeof state === "object" ? state.error : null;
                            return (
                                <li
                                    key={devServerKey(server)}
                                    data-dev-server-row=""
                                    className="px-3 py-1.5">
                                    <div className="flex items-center gap-2">
                                        <span className="bg-success h-2 w-2 shrink-0 rounded-sm" />
                                        <div className="min-w-0 flex-1">
                                            <p className="text-foreground truncate text-sm">
                                                {server.label}
                                            </p>
                                            <p className="text-muted-foreground text-xs">
                                                pid {server.pid} ·{" "}
                                                {formatRelativeTime(server.registeredAt)}
                                            </p>
                                        </div>
                                        <Button
                                            variant="outline"
                                            size="xs"
                                            disabled={stopping}
                                            onClick={() => void stopDevServer(server)}
                                            className="text-destructive border-destructive/40 shrink-0 [-webkit-app-region:no-drag]">
                                            {stopping ? "Stopping…" : "Stop"}
                                        </Button>
                                    </div>
                                    {error && (
                                        <p className="text-destructive mt-1 pl-4 text-xs">
                                            {error}
                                        </p>
                                    )}
                                </li>
                            );
                        })}
                    </ul>
                </div>
            ))}
        </div>
    );
}

function DevServerButton() {
    const [open, setOpen] = useState(false);
    const count = useDevServerStore((s) => s.devServers.length);
    const clearStopErrors = useDevServerStore((s) => s.clearStopErrors);

    const handleOpenChange = useCallback(
        (next: boolean) => {
            setOpen(next);
            if (!next) clearStopErrors();
        },
        [clearStopErrors],
    );

    // Stays while open, so stopping the last server doesn't pull the popover's anchor away.
    if (count === 0 && !open) return null;

    return (
        <Popover open={open} onOpenChange={handleOpenChange}>
            <PopoverTrigger asChild>
                <Button
                    variant="ghost"
                    size="icon-xs"
                    aria-label="Dev servers"
                    tooltip="Dev servers"
                    tooltipSide="right"
                    className="relative [-webkit-app-region:no-drag]">
                    <Server className="h-3.5 w-3.5" />
                    {count > 0 && (
                        <span className="bg-success text-success-foreground absolute -top-0.5 -right-0.5 flex h-3 min-w-3 items-center justify-center rounded-full px-0.5 text-[8px] font-medium">
                            {count}
                        </span>
                    )}
                </Button>
            </PopoverTrigger>
            <PopoverContent side="right" align="end" sideOffset={8} className="w-80 p-0">
                <div className="border-border border-b px-3 py-2">
                    <span className="text-sm font-medium">Running dev servers</span>
                </div>
                <DevServerList />
            </PopoverContent>
        </Popover>
    );
}

export { DevServerButton, DevServerList };
```
